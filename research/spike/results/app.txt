CREATE TABLE products (
    id     bigint        PRIMARY KEY,
    sku    text          NOT NULL UNIQUE,
    name   text          NOT NULL,
    price  numeric(12,2) NOT NULL CHECK (price >= 0),
    stock  integer       NOT NULL DEFAULT 0,
    attrs  jsonb         NOT NULL DEFAULT '{}'
);
CREATE TABLE
CREATE TABLE orders (
    id           bigint        PRIMARY KEY,
    customer_id  uuid          NOT NULL REFERENCES customers (id),

    placed_at    timestamptz   NOT NULL DEFAULT now()
);

CREATE TABLE order_items (
    order_id    bigint        NOT NULL REFERENCES orders (id) ON DELETE CASCADE,

    unit_price  numeric(12,2) NOT NULL,
    PRIMARY KEY (order_id, product_id)
);

Q01 join + aggregate: revenue per customer
SELECT c.name, count(o.id) AS orders, coalesce(sum(o.total), 0) AS revenue
FROM customers c LEFT JOIN orders o ON o.customer_id = c.id
GROUP BY c.name
ORDER BY revenue DESC, c.name;

Q02 three-way join + ORDER BY + LIMIT
SELECT o.id, c.email, p.sku, oi.qty, oi.qty * oi.unit_price AS line_total
FROM orders o
JOIN customers c ON c.id = o.customer_id
JOIN order_items oi ON oi.order_id = o.id
